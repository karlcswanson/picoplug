# Host-side unit tests for the LLDP parser.
#   cmake -S firmware/test -B build-test && cmake --build build-test && ctest --test-dir build-test
#
# Sanitizers: UBSan is on by default. ASan is on by default on Linux only;
# the ASan runtime in Apple clang 15 and Homebrew LLVM 20 hangs or aborts at
# startup on macOS 26+, so it is off there. Out-of-bounds reads are still
# caught on every platform: the fuzz-style tests place each input buffer
# directly before an mmap PROT_NONE guard page.
cmake_minimum_required(VERSION 3.16)
project(lldp_tests C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(LLDP_TEST_UBSAN "Build tests with UBSan" ON)
if(APPLE)
    option(LLDP_TEST_ASAN "Build tests with ASan" OFF)
else()
    option(LLDP_TEST_ASAN "Build tests with ASan" ON)
endif()

add_executable(test_lldp test_lldp.c ../src/lldp.c)
target_include_directories(test_lldp PRIVATE ../src)
target_compile_definitions(test_lldp PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
target_compile_options(test_lldp PRIVATE -Wall -Wextra -Wconversion -Wshadow -Wvla -pedantic -O1 -g)

set(_san "")
if(LLDP_TEST_ASAN)
    list(APPEND _san address)
endif()
if(LLDP_TEST_UBSAN)
    list(APPEND _san undefined)
endif()
if(_san)
    list(JOIN _san "," _san_list)
    target_compile_options(test_lldp PRIVATE -fsanitize=${_san_list} -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(test_lldp PRIVATE -fsanitize=${_san_list})
endif()

enable_testing()
foreach(t iterate decode get)
    add_test(NAME query.${t} COMMAND test_query ${t})
endforeach()
foreach(t parse bounds tracker)
    add_test(NAME eapol.${t} COMMAND test_eapol ${t})
endforeach()
foreach(t parse bounds tracker)
    add_test(NAME lacp.${t} COMMAND test_lacp ${t})
endforeach()
foreach(t roundtrip bounds table verdict)
    add_test(NAME peer.${t} COMMAND test_peer ${t})
endforeach()
foreach(t direct switch dead responder refusals)
    add_test(NAME path.${t} COMMAND test_path ${t})
endforeach()
foreach(t igmp ptp bounds tracker)
    add_test(NAME mcast.${t} COMMAND test_mcast ${t})
endforeach()
foreach(t roundtrip query)
    add_test(NAME settings.${t} COMMAND test_settings ${t})
endforeach()
foreach(t config rstp tcn pvst errors tracker)
    add_test(NAME stp.${t} COMMAND test_stp ${t})
endforeach()
foreach(t ack flow bounds)
    add_test(NAME dhcp_info.${t} COMMAND test_dhcp_info ${t})
endforeach()
foreach(t lldp_roundtrip lldp_shutdown lldp_minimal lldp_bounds cdp_roundtrip cdp_minimal cdp_checksum_quirk cdp_bounds)
    add_test(NAME tx.${t} COMMAND test_tx ${t})
endforeach()
foreach(t switch phone router v1 errors truncation mutation random)
    add_test(NAME cdp.${t} COMMAND test_cdp ${t})
endforeach()
add_test(NAME json.writer COMMAND test_json writer)
add_test(NAME json.status COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/check_json.py $<TARGET_FILE:test_json>)

add_test(NAME page.smoke COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/test_page.py)
set_tests_properties(page.smoke PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120)

    add_test(NAME lldp.${t} COMMAND test_lldp ${t})
endforeach()
